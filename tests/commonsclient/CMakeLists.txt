cmake_minimum_required(VERSION 3.24)
project(CommonsClientTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Test)
enable_testing()
add_executable(commonsclienttest commonsclienttest.cpp ../../src/assistant/commonsclient.cpp ../../src/assistant/workspacepaths.cpp)
target_link_libraries(commonsclienttest PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME commonsclienttest COMMAND commonsclienttest)
set_tests_properties(commonsclienttest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

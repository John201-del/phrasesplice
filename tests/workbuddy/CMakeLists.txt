cmake_minimum_required(VERSION 3.20)
project(LocalCutWorkBuddyTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
add_executable(workbuddybridgetest workbuddybridgetest.cpp
    ../../src/assistant/workbuddybridge.cpp ../../src/assistant/workspacepaths.cpp)
target_link_libraries(workbuddybridgetest PRIVATE Qt6::Core Qt6::Network Qt6::Test)
target_compile_definitions(workbuddybridgetest PRIVATE QT_USE_QSTRINGBUILDER)
if(WIN32)
    target_link_libraries(workbuddybridgetest PRIVATE advapi32)
endif()
add_executable(localcut-mcp ../../src/assistant/mcpbridge_main.cpp)
target_link_libraries(localcut-mcp PRIVATE Qt6::Core Qt6::Network)
add_executable(workbuddy-output-sink output_sink.cpp)
target_link_libraries(workbuddy-output-sink PRIVATE Qt6::Core)
add_dependencies(workbuddybridgetest localcut-mcp workbuddy-output-sink)
if(MSVC)
    target_compile_options(workbuddybridgetest PRIVATE /utf-8)
    target_compile_options(localcut-mcp PRIVATE /utf-8)
endif()
add_test(NAME workbuddybridgetest COMMAND workbuddybridgetest)

cmake_minimum_required(VERSION 3.20)
project(MediaTasksTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
enable_testing()
add_executable(media-slow-tool slowtool.cpp)
target_link_libraries(media-slow-tool PRIVATE Qt6::Core)
add_executable(mediataskstest mediataskstest.cpp ../../src/assistant/mediatasks.cpp ../../src/assistant/workspacepaths.cpp)
add_dependencies(mediataskstest media-slow-tool)
target_include_directories(mediataskstest PRIVATE ../../src)
target_compile_definitions(mediataskstest PRIVATE QT_NO_KEYWORDS)
target_link_libraries(mediataskstest PRIVATE Qt6::Core Qt6::Test)
add_test(NAME mediataskstest COMMAND mediataskstest)

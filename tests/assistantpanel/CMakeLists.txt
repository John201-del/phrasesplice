cmake_minimum_required(VERSION 3.24)
project(AssistantPanelTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network Test)
enable_testing()
add_executable(assistantpaneltest assistantpaneltest.cpp ../../src/assistant/assistantpanel.cpp ../../src/assistant/workspacepaths.cpp)
target_link_libraries(assistantpaneltest PRIVATE Qt6::Core Qt6::Widgets Qt6::Test)
add_test(NAME assistantpaneltest COMMAND assistantpaneltest)
set_tests_properties(assistantpaneltest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(assistantconnectiondialogtest assistantconnectiondialogtest.cpp ../../src/assistant/assistantconnectiondialog.cpp ../../src/assistant/workspacepaths.cpp)
target_link_libraries(assistantconnectiondialogtest PRIVATE Qt6::Core Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME assistantconnectiondialogtest COMMAND assistantconnectiondialogtest)
set_tests_properties(assistantconnectiondialogtest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

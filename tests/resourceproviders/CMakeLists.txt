cmake_minimum_required(VERSION 3.20)
project(ResourceProviderTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network NetworkAuth Test)
find_package(KF6Config REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(KF6WidgetsAddons REQUIRED)
find_package(KF6KIO REQUIRED)
add_executable(resourceprovidertest test.cpp ../../src/onlineresources/providermodel.cpp)
target_include_directories(resourceprovidertest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(resourceprovidertest PRIVATE QT_USE_QSTRINGBUILDER)
target_link_libraries(resourceprovidertest PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::NetworkAuth Qt6::Test
    KF6::ConfigCore KF6::I18n KF6::WidgetsAddons KF6::KIOCore)
if(MSVC)
    target_compile_options(resourceprovidertest PRIVATE /utf-8)
endif()
add_test(NAME resourceprovidertest COMMAND resourceprovidertest)
